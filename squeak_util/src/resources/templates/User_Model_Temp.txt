<?php
namespace models;

class User_Model {
    public function create_user($user_data, $db) {
        $hash_gen = new \utils\Hash_Gen();
        $salt = $hash_gen->salt();
        $join_date = floor(microtime(true) * 1000);
        $hash = $hash_gen->hash($user_data["password"], $salt);

        $query = "INSERT INTO users (username, join_date, salt, hash) VALUES (:username, :join_date, :salt, :hash)";

        $params = [
            [
                "name" => ":username",
                "value" => $user_data["username"],
                "type" => "s"
            ],
            [
                "name" => ":join_date",
                "value" => $join_date,
                "type" => "i"
            ],
            [
                "name" => ":salt",
                "value" => $salt,
                "type" => "s"
            ],
            [
                "name" => ":hash",
                "value" => $hash,
                "type" => "s"
            ],
        ];

        try {
            $out = $db->make_query("insert", $query, $params);
            if($out) {
                return true;
            }
            return false;
        }
        catch(\Exception $e) {
            return false;
        }
    }

    public function get_user_by_name($username, $db) {
        $query = "SELECT * FROM users WHERE username = :username";
        $params = [
            [
                "name" => ":username",
                "value" => $username,
                "type" => "s"
            ]
        ];

        try {
            return $db->make_query("select", $query, $params);
        }
        catch(\Exception $e) {
            return false;
        }
    }

    public function get_user_by_id($user_id, $db) {
        $query = "SELECT (id, username, join_date) FROM users WHERE id = :user_id";
        $params = [
            [
                "name" => ":user_id",
                "value" => $user_id,
                "type" => "i"
            ]
        ];

        try {
            return $db->make_query("select", $query, $params);
        }
        catch(\Exception $e) {
            return false;
        }
    }

    public function get_session_token($user_id, $db) {
        $query = "SELECT * FROM session_token WHERE user_id = :user_id";
        $params = [
            [
                "name" => ":user_id",
                "value" => $user_id,
                "type" => "i"
            ]
        ];
        try {
            $resp = $db->make_query("select", $query, $params);
            if($resp && sizeof($resp) > 0) {
                return $resp[0]["token"];
            }
            return false;
        }
        catch(\Exception $e) {
            return false;
        }
    }

    public function create_session($user_id, $db) {
        $hash_gen = new \utils\Hash_Gen();
        $token = $hash_gen->salt(32);

        $query = "REPLACE INTO session_token (user_id, token, exp) VALUES (:user_id, :token, :exp)";
        $params = [
            [
                "name" => ":user_id",
                "value" => $user_id,
                "type" => "i"
            ],
            [
                "name" => ":token",
                "value" => $token,
                "type" => "s"
            ],
            [
                "name" => ":exp",
                "value" => (int) round(microtime(true) * 1000) + 600000,
                "type" => "i"
            ]
        ];

        try {
            $out = $db->make_query("replace", $query, $params);
            if(!$out) {
                return false;
            }
            return $token;
        }
        catch(\Exception $e) {
            return false;
        }
    }

        $query = "REPLACE INTO kli (user_id, token) VALUES (:user_id, :token)";
        $params = [
            [
                "name" => ":user_id",
                "value" => $user_id,
                "type" => "i"
            ],
            [
                "name" => ":token",
                "value" => $hash,
                "type" => "s"
            ]
        ];

        try {
            $db->make_query("replace", $query, $params);

            return $token;
        }
        catch(\Exception $e) {

    }

    public function get_kli($token, $db) {
        $hash_gen = new \utils\Hash_Gen();
        $hash = $hash_gen->hash($token, "squeak");

        $query = "SELECT * FROM kli WHERE token = :token";
        $params = [
            [
                "name" => ":token",
                "value" => $hash,
                "type" => "s"
            ]
        ];

        try {
            $resp = $db->make_query("select", $query, $params);
            if($resp && sizeof($resp) > 0) {
                return $resp[0];
            }
            return false;
        }
        catch(\Exception $e) {
            return false;
        }
    }

    public function clear_session($user_id, $db) {
        $query = "DELETE FROM session_token WHERE user_id = :user_id";
        $params = [
            [
                "name" => ":user_id",
                "value" => $user_id,
                "type" => "i"
            ]
        ];
        try {
            $db->make_query("delete", $query, $params);
            return true;
        }
        catch(\Exception $e) {
            return false;
        }
    }

    public function clear_kli($user_id, $db) {
        $query = "DELETE FROM kli WHERE user_id = :user_id";
        $params = [
            [
                "name" => ":user_id",
                "value" => $user_id,
                "type" => "i"
            ]
        ];
